= Start_Date + End_Date
Ridership Query
= Dataset_ID + Start_Date + End_Date
Ridership Records
= 1{Service_Date + Day_Type + Bus_Rides + Rail_Rides + Total_Rides}N
Raw Ridership
= Ingest_Date + 1{Service_Date + Day_Type + Bus_Rides + Rail_Rides + Total_Rides}N
Weather Query
= Latitude + Longitude + Start_Date + End_Date + Daily_Variables
Weather Observations
= 1{Weather_Date + Temp_Max_C + Temp_Min_C + Precipitation_mm + Snowfall_cm}N
Raw Weather
= Ingest_Date + 1{Weather_Date + Temp_Max_C + Temp_Min_C + Precipitation_mm + Snowfall_cm}N
Clean Ridership
= 1{Service_Date + Day_Type + Bus_Rides + Rail_Rides + Total_Rides}N
Clean Weather
= 1{Weather_Date + Temp_Max_C + Temp_Min_C + Precipitation_mm + Snowfall_cm}N
Holiday List
= 1{Holiday_Date + Holiday_Name}N
Cleaning Report
= Rows_Loaded + Rows_Invalid + Rows_Duplicate + Rows_Kept
Merged Daily Record
= Service_Date + Total_Rides + Bus_Rides + Rail_Rides + Temp_Max_C + Temp_Min_C + Precipitation_mm + Snowfall_cm + Weekday + Month + Is_Weekend + Is_Holiday

Phase 6 · Multilayer sequence diagram: daily ingest

UML sequence diagram

Object lifelines (name:Class), messages written as returnValue := message(parameters), a dashed return after each synchronous call, alt frames whose guard conditions Mermaid shows in brackets, «create» for each new raw file, and assumptions in a note.

sequenceDiagram
    actor SCH as :Scheduler
    participant H as :IngestHandler
    participant PS as :ParameterStore
    participant CDP as :ChicagoDataPortal
    participant OM as :OpenMeteo
    participant S3 as :S3DataAccess
    participant CW as :CloudWatch
    Note over H,CDP: Assumptions: targetDate defaults to yesterday. Athena reads clean/ later through Glue tables, outside this run.
    SCH->>H: summary := handler(event, context)
    H->>PS: config := get_config()
    PS-->>H: config
    H->>CDP: rides := fetch_ridership(targetDate, targetDate)
    CDP-->>H: rides
    alt rides returned
        create participant RR as ridershipFile:RawFile
        H->>RR: «create» RawFile(ridership, targetDate, rides)
        H->>S3: rideKey := upload_raw(bucket, ridership, targetDate, rides)
        S3-->>H: rideKey
    else error or no data yet
        H-)CW: putMetric(IngestFailure, 1)
    end
    H->>OM: weather := fetch_weather(targetDate, targetDate, lat, lon)
    OM-->>H: weather
    alt weather returned
        create participant RW as weatherFile:RawFile
        H->>RW: «create» RawFile(weather, targetDate, weather)
        H->>S3: weatherKey := upload_raw(bucket, weather, targetDate, weather)
        S3-->>H: weatherKey
    else error
        H-)CW: putMetric(IngestFailure, 1)
    end
    H->>S3: cleanKey := write_parquet(bucket, clean/, cleanDf)
    S3-->>H: cleanKey
    H-)CW: logSummary(targetDate, rowsWritten)
